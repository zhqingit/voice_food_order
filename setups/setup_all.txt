# Rate limits (this file is included inside nginx's http{} block, so zones go here).
# auth: login/signup/guest/refresh per client IP; voice_conn: concurrent voice sockets per IP.
  limit_req_zone $binary_remote_addr zone=auth:10m rate=30r/m;
  limit_conn_zone $binary_remote_addr zone=voice_conn:10m;
  limit_req_status 429;
  limit_conn_status 429;

# Voxeats — store portal + API
  server {
      listen 443 ssl;
      server_name voxeats.rest-oai.com;

      ssl_certificate /etc/letsencrypt/live/voxeats.rest-oai.com/fullchain.pem;
      ssl_certificate_key /etc/letsencrypt/live/voxeats.rest-oai.com/privkey.pem;
      include /etc/letsencrypt/options-ssl-nginx.conf;
      ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

      # Store portal — static SPA at root
      root /var/www/store-portal;
      index index.html;

      location / {
          try_files $uri $uri/ /index.html;
      }

      # Auth endpoints — rate limited (guest signup creates a user per call)
      location /store/auth/ {
          limit_req zone=auth burst=20 nodelay;
          proxy_pass http://127.0.0.1:8000;
          proxy_set_header Host store-api.local;
          proxy_set_header X-Real-IP $remote_addr;
      }

      location /user/auth/ {
          limit_req zone=auth burst=20 nodelay;
          proxy_pass http://127.0.0.1:8000;
          proxy_set_header Host user-api.local;
          proxy_set_header X-Real-IP $remote_addr;
      }

      # Store API
      location /store/ {
          proxy_pass http://127.0.0.1:8000;
          proxy_set_header Host store-api.local;
          proxy_set_header X-Real-IP $remote_addr;
      }

      # User API (mobile app)
      location /user/ {
          proxy_pass http://127.0.0.1:8000;
          proxy_set_header Host user-api.local;
          proxy_set_header X-Real-IP $remote_addr;
      }

      # Voice WebSocket — each open socket is a paid Gemini Live session
      location /voice/ws {
          limit_conn voice_conn 2;
          proxy_pass http://127.0.0.1:8000;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection "upgrade";
          proxy_set_header Host user-api.local;
          proxy_read_timeout 3600s;
      }

      # Uploaded store logos (validated PNG/JPEG/WebP only)
      location /uploads/ {
          proxy_pass http://127.0.0.1:8000;
          add_header X-Content-Type-Options nosniff;
      }

      # Voice REST (sessions, orders)
      location /voice/ {
          proxy_pass http://127.0.0.1:8000;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection "upgrade";
          proxy_set_header Host user-api.local;
          proxy_read_timeout 3600s;
      }
  }

  # HTTP → HTTPS redirect for voxeats
  server {
      listen 80;
      server_name voxeats.rest-oai.com;
      return 301 https://$host$request_uri;
  }

  # Company website
  server {
      server_name www.rest-oai.com rest-oai.com;

      root /var/www/restoai;
      index index.html;

      location / {
          try_files $uri $uri/ /index.html;
      }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/www.rest-oai.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/www.rest-oai.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}

    if ($host = rest-oai.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = www.rest-oai.com) {
